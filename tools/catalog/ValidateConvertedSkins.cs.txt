var output=OUTPUT;
if(UnityEngine.AssetBundle.GetAllLoadedAssetBundles().Any())throw new System.Exception("Skin fixture requires no existing bundle owners");
var directory=System.IO.Path.GetDirectoryName(output);
var requestPath=(string)Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(System.IO.Path.Combine(directory,"inputs.json")))["request"];
var request=Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(requestPath));
System.Func<Newtonsoft.Json.Linq.JToken,string> pin=row=>{
 var path=System.IO.Path.GetFullPath(System.IO.Path.Combine(System.IO.Path.GetDirectoryName(requestPath),(string)row["path"]));
 using(var stream=System.IO.File.OpenRead(path))using(var sha=System.Security.Cryptography.SHA256.Create()){
  var hash=System.BitConverter.ToString(sha.ComputeHash(stream)).Replace("-","").ToLowerInvariant();
  if(stream.Length!=(long)row["bytes"]||hash!=(string)row["sha256"])throw new System.Exception("Skin fixture input pin differs: "+path);
 }return path;
};
var manifest=Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(pin(request["metadata"])));
var locations=manifest["locations"].ToDictionary(row=>(int)row["id"]);
var stringKeys = manifest["keys"]
    .Where(row => (string)row["keyType"] == "System.String" || row["keyType"] == null)
    .ToDictionary(row => (string)row["key"], row => row["locations"].Select(id => locations[(int)id]).ToArray());
var fixture=request["skin_fixture"];
var pins=(Newtonsoft.Json.Linq.JObject)request["bundles"];
var startupOwners=fixture==null?pins.Properties().Select(row=>row.Name).ToArray():new[]{"whole-game-content","commando-prefab-lab","objective-support-lab"};
var bundles=new System.Collections.Generic.Dictionary<string,UnityEngine.AssetBundle>();
var rows=new System.Collections.Generic.List<object>();var failures=new System.Collections.Generic.List<object>();
var typedBodies=new System.Collections.Generic.List<System.Collections.Generic.KeyValuePair<string,UnityEngine.GameObject>>();
int bodies=0,controllers=0,slots=0;
int checkedTransforms = 0;
var checkedSkins = new System.Collections.Generic.HashSet<RoR2.SkinDef>();
var projectileConfigPin = request["projectile_config"];
var checkedProjectileKeys = new System.Collections.Generic.HashSet<string>();

System.Func<string, UnityEngine.AssetBundle> openBundle = owner =>
{
    if (bundles.TryGetValue(owner, out var bundle))
        return bundle;
    bundle = UnityEngine.AssetBundle.LoadFromFile(pin(pins[owner]));
    if (!bundle)
        throw new System.Exception("Skin fixture bundle failed to load: " + owner);
    bundles.Add(owner, bundle);
    return bundle;
};

System.Func<UnityEngine.Transform, object> describeTransform = transform =>
{
    if (!transform)
        return null;
    var path = new System.Collections.Generic.Stack<string>();
    for (var current = transform; current; current = current.parent)
        path.Push(current.name);
    return new { path = string.Join("/", path), instanceId = transform.GetInstanceID(),
                 prefabInstanceId = transform.root.gameObject.GetInstanceID() };
};

System.Action<RoR2.SkinDef, string, string, int, UnityEngine.Transform> checkSkin = null;
checkSkin = (skin, body, path, index, model) =>
{
    if (!skin || !checkedSkins.Add(skin))
        return;
    foreach (var baseSkin in skin.baseSkins)
        checkSkin(baseSkin, body, path, index, model);
    var reference = skin.GetSkinParams();
    var parameters = reference.directRef;
    var key = reference.assetRef?.RuntimeKey as string;
    Newtonsoft.Json.Linq.JToken location = null;
    try
    {
        if (reference.assetRef != null && reference.assetRef.RuntimeKeyIsValid())
        {
            if (key == null || !stringKeys.TryGetValue(key, out var candidates))
                throw new System.Exception("SkinDefParams address absent");
            location = candidates.FirstOrDefault(row =>
                ((string)row["type"])?.Split(',')[0] == typeof(RoR2.SkinDefParams).FullName);
            if (location == null)
                throw new System.Exception("Typed SkinDefParams location absent");
            parameters = openBundle((string)location["bundle"])
                .LoadAsset<RoR2.SkinDefParams>((string)location["path"]);
        }
        if (!parameters)
            throw new System.Exception("SkinDefParams object missing");
    }
    catch (System.Exception error)
    {
        failures.Add(new { body, path, index, skin = skin.name, key, location, error = error.Message });
        return;
    }
    var root = skin.rootObject ? skin.rootObject.transform : null;
    System.Action<string, int, UnityEngine.Object, UnityEngine.Transform> checkTransform = (field, element, target, transform) =>
    {
        checkedTransforms++;
        try
        {
            RoR2.Util.BuildPrefabTransformPath(root, transform);
        }
        catch (System.ArgumentException error)
        {
            failures.Add(new { body, path, index, skin = skin.name, key, location, field, element,
                root = describeTransform(root), model = describeTransform(model),
                target = describeTransform(transform), targetInstanceId = target ? target.GetInstanceID() : 0,
                error = error.Message });
        }
    };
    for (int i = 0; i < parameters.rendererInfos.Length; i++)
    {
        var renderer = parameters.rendererInfos[i].renderer;
        checkTransform("rendererInfos", i, renderer, renderer ? renderer.transform : null);
    }
    for (int i = 0; i < parameters.gameObjectActivations.Length; i++)
    {
        var gameObject = parameters.gameObjectActivations[i].gameObject;
        checkTransform("gameObjectActivations", i, gameObject, gameObject ? gameObject.transform : null);
    }
    for (int i = 0; i < parameters.meshReplacements.Length; i++)
    {
        var renderer = parameters.meshReplacements[i].renderer;
        checkTransform("meshReplacements", i, renderer, renderer ? renderer.transform : null);
    }
    for (int i = 0; i < parameters.lightReplacements.Length; i++)
    {
        var light = parameters.lightReplacements[i].light;
        checkTransform("lightReplacements", i, light, light ? light.transform : null);
    }
};

System.Func<string,string,UnityEngine.GameObject> loadBody=(owner,path)=>{
 var body=bundles[owner].LoadAsset<UnityEngine.GameObject>(path);
 if(!body)throw new System.Exception("Typed body missing: "+owner+"/"+path);
 return body;
};
try{
 foreach(var owner in startupOwners)openBundle(owner);
 // Earlier startup orders are retained as controls for first-load failures.
 var wardPath=(string)fixture?["ward_path"];
 var commandoPath=(string)fixture?["commando_path"];
 bool earlyWard=(bool?)fixture?["early_ward"]??false;
 bool earlyCommando=(bool?)fixture?["early_commando"]??false;
 if(earlyWard)loadBody("objective-support-lab",wardPath);
 if(earlyCommando)loadBody("commando-prefab-lab",commandoPath);
 var groups=WholeCatalogComposed.FamilyLocationGroups(manifest,"CharacterBody");
 foreach(int id in groups.SelectMany(group=>group).Distinct()){
  var row=locations[id];string path=(string)row["path"],owner=(string)row["bundle"];
  try{
   typedBodies.Add(new System.Collections.Generic.KeyValuePair<string,UnityEngine.GameObject>(path,loadBody(owner,path)));
  }catch(System.Exception error){failures.Add(new{path,owner,error=error.ToString()});}
 }
 if(commandoPath!=null&&!earlyCommando)loadBody("commando-prefab-lab",commandoPath);
 if(wardPath!=null&&!earlyWard)loadBody("objective-support-lab",wardPath);
 foreach(var entry in typedBodies){
  var path=entry.Key;var body=entry.Value;
  try{
   bodies++;
   var locator=body.GetComponent<RoR2.ModelLocator>();
   var model=locator?locator.modelTransform:null;
   var liveController=model?model.GetComponent<RoR2.ModelSkinController>():null;
   var bodyControllers=fixture==null?body.GetComponentsInChildren<RoR2.ModelSkinController>(true):liveController?new[]{liveController}:new RoR2.ModelSkinController[0];
   foreach(var controller in bodyControllers){
    controllers++;
    if(controller.skins==null)throw new System.Exception("skins array null");
    for(int index=0;index<controller.skins.Length;index++){
     var skin=controller.skins[index];slots++;
     var expected=manifest["bodySkinTargets"]?[body.name]?[index.ToString(System.Globalization.CultureInfo.InvariantCulture)];
     rows.Add(new{body=body.name,path,index,loaded=(bool)skin,skin=skin?skin.name:null,root=skin&&skin.rootObject?skin.rootObject.name:null,expected});
     if(!skin)failures.Add(new{body=body.name,path,index,expected,error="ModelSkinController.skins slot null"});
     else if(skin.GetSkinParams().assetRef==null)failures.Add(new{body=body.name,path,index,expected,error="SkinDef.GetSkinParams().assetRef null"});
     else checkSkin(skin, body.name, path, index, model);
    }
   }
  }catch(System.Exception error){failures.Add(new{path,error=error.ToString()});}
 }
 if (projectileConfigPin != null)
 {
     // Inspect the same objective config fields that prepare native projectiles on Start.
     // Load these after the skin checks to preserve their first-deserialization order.
     var config = Newtonsoft.Json.Linq.JObject.Parse(System.IO.File.ReadAllText(pin(projectileConfigPin)));
     var visited = new System.Collections.Generic.HashSet<UnityEngine.GameObject>();
     System.Action<UnityEngine.GameObject, string, string> checkProjectiles = null;
     checkProjectiles = (root, path, field) =>
     {
         if (!root || !visited.Add(root))
             return;
         foreach (var trail in root.GetComponentsInChildren<RoR2.DamageTrail>(true))
             checkProjectiles(trail.segmentPrefab, path, field);
         foreach (var projectile in root.GetComponentsInChildren<RoR2.Projectile.ProjectileController>(true))
         {
             var reference = projectile.ghostPrefabAddress;
             if (reference == null || !reference.RuntimeKeyIsValid())
             {
                 checkProjectiles(projectile.ghostPrefab, path, field);
                 continue;
             }
             var key = reference.RuntimeKey.ToString();
             if (!checkedProjectileKeys.Add(key))
                 continue;
             Newtonsoft.Json.Linq.JToken location = null;
             try
             {
                 if (stringKeys.TryGetValue(key, out var candidates))
                     location = candidates.FirstOrDefault(row =>
                         ((string)row["type"])?.Split(',')[0] == typeof(UnityEngine.GameObject).FullName);
                 if (location == null)
                     throw new System.Exception("Typed projectile ghost address absent");
                 var ghost = openBundle((string)location["bundle"])
                     .LoadAsset<UnityEngine.GameObject>((string)location["path"]);
                 if (!ghost || !ghost.GetComponent<RoR2.Projectile.ProjectileGhostController>() ||
                     !ghost.GetComponentsInChildren<UnityEngine.Component>(true).All(component => component))
                     throw new System.Exception("Typed projectile ghost component contract failed");
                 checkProjectiles(ghost, path, field);
             }
             catch (System.Exception error)
             {
                 failures.Add(new { path, field, projectile = projectile.name, key, location, error = error.Message });
             }
         }
     };
     var owner = openBundle("commando-prefab-lab");
     foreach (var asset in config["objectiveConfigAssets"])
     {
         var path = (string)asset;
         var source = owner.LoadAsset<RoR2.EntityStateConfiguration>(path);
         if (!source)
         {
             failures.Add(new { path, error = "Typed objective state config missing" });
             continue;
         }
         foreach (var field in source.serializedFieldsCollection.serializedFields)
         {
             var projectile = field.fieldValue.objectValue as UnityEngine.GameObject;
             if (projectile && projectile.GetComponent<RoR2.Projectile.ProjectileController>())
                 checkProjectiles(projectile, path, field.fieldName);
         }
     }
 }
}catch(System.Exception error){failures.Add(new{error=error.ToString()});}
finally{foreach(var bundle in bundles.Values.Reverse())bundle.Unload(true);}
var summary=new{bodies,controllers,slots,checkedSkins=checkedSkins.Count,checkedTransforms,checkedProjectileKeys=checkedProjectileKeys.Count,failed=failures.Count,scope="Exact-editor typed converted bundle references and skin bake hierarchy"};
var result=new{success=failures.Count==0&&bodies>0,summary,startupOwners,deferredOwners=pins.Properties().Select(row=>row.Name).Except(startupOwners),fixture,rows,failures};
System.IO.File.WriteAllText(output,Newtonsoft.Json.JsonConvert.SerializeObject(result,Newtonsoft.Json.Formatting.Indented));return summary;
