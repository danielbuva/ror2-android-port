 using(var stream=System.IO.File.OpenRead(path))using(var sha=System.Security.Cryptography.SHA256.Create()){

  if(stream.Length!=(long)row["bytes"]||hash!=(string)row["sha256"])throw new System.Exception("Skin fixture input pin differs: "+path);
 }return path;
};

 }
