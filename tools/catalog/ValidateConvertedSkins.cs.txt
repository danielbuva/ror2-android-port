System.Func<string, UnityEngine.AssetBundle> openBundle = owner =>
{
    if (bundles.TryGetValue(owner, out var bundle))
        return bundle;
    bundle = UnityEngine.AssetBundle.LoadFromFile(pin(pins[owner]));

        throw new System.Exception("Skin fixture bundle failed to load: " + owner);

System.Func<UnityEngine.Transform, object> describeTransform = transform =>
{
    if (!transform)
        return null;
    var path = new System.Collections.Generic.Stack<string>();
    for (var current = transform; current; current = current.parent)
        path.Push(current.name);
    return new { path = string.Join("/", path), instanceId = transform.GetInstanceID(),
                 prefabInstanceId = transform.root.gameObject.GetInstanceID() };
};

System.Action<RoR2.SkinDef, string, string, int, UnityEngine.Transform> checkSkin = null;
checkSkin = (skin, body, path, index, model) =>
{
    if (!skin || !checkedSkins.Add(skin))
        return;
    foreach (var baseSkin in skin.baseSkins)
        checkSkin(baseSkin, body, path, index, model);
    var reference = skin.GetSkinParams();
    var parameters = reference.directRef;
    var key = reference.assetRef?.RuntimeKey as string;
    Newtonsoft.Json.Linq.JToken location = null;
    try
    {
        if (reference.assetRef != null && reference.assetRef.RuntimeKeyIsValid())
        {
            if (key == null || !stringKeys.TryGetValue(key, out var candidates))
                throw new System.Exception("SkinDefParams address absent");
            location = candidates.FirstOrDefault(row =>
                ((string)row["type"])?.Split(',')[0] == typeof(RoR2.SkinDefParams).FullName);
            if (location == null)
                throw new System.Exception("Typed SkinDefParams location absent");
            parameters = openBundle((string)location["bundle"])
                .LoadAsset<RoR2.SkinDefParams>((string)location["path"]);
        }
        if (!parameters)
            throw new System.Exception("SkinDefParams object missing");
    }
    catch (System.Exception error)
    {
        failures.Add(new { body, path, index, skin = skin.name, key, location, error = error.Message });
        return;
    }
    var root = skin.rootObject ? skin.rootObject.transform : null;
    System.Action<string, int, UnityEngine.Object, UnityEngine.Transform> checkTransform = (field, element, target, transform) =>
    {
        checkedTransforms++;
        try
        {
            RoR2.Util.BuildPrefabTransformPath(root, transform);
        }
        catch (System.ArgumentException error)
        {
            failures.Add(new { body, path, index, skin = skin.name, key, location, field, element,
                root = describeTransform(root), model = describeTransform(model),
                target = describeTransform(transform), targetInstanceId = target ? target.GetInstanceID() : 0,
                error = error.Message });
        }
    };
    for (int i = 0; i < parameters.rendererInfos.Length; i++)
    {
        var renderer = parameters.rendererInfos[i].renderer;
        checkTransform("rendererInfos", i, renderer, renderer ? renderer.transform : null);
    }
    for (int i = 0; i < parameters.gameObjectActivations.Length; i++)
    {
        var gameObject = parameters.gameObjectActivations[i].gameObject;
        checkTransform("gameObjectActivations", i, gameObject, gameObject ? gameObject.transform : null);
    }
    for (int i = 0; i < parameters.meshReplacements.Length; i++)
    {
        var renderer = parameters.meshReplacements[i].renderer;
        checkTransform("meshReplacements", i, renderer, renderer ? renderer.transform : null);
    }
    for (int i = 0; i < parameters.lightReplacements.Length; i++)
    {
        var light = parameters.lightReplacements[i].light;
        checkTransform("lightReplacements", i, light, light ? light.transform : null);
    }
};

         var badPairs = relationships?.Select((pair, index) => new { index, pair.itemDef1, pair.itemDef2 })
             .Where(pair => !itemDefs.Contains(pair.itemDef1) || !itemDefs.Contains(pair.itemDef2))
             .Select(pair => new { pair.index, first = pair.itemDef1 ? pair.itemDef1.name : null,
                 second = pair.itemDef2 ? pair.itemDef2.name : null }).ToArray();
         if (!provider.relationshipType || !relationshipTypes.Contains(provider.relationshipType) || badPairs == null || badPairs.Length != 0)
             failDefinition(new { name = provider.name, relationshipType = provider.relationshipType ? provider.relationshipType.name : null,
                 pairs = relationships?.Length, badPairs, error = "Item relationship points outside the typed definition families" });
     }
     definitionsComplete = true;

     var sourceBodyIds = WholeCatalogComposed.FamilyLocationGroups(source, "CharacterBody").SelectMany(group => group).Distinct().ToArray();
     var missingBodyIds = sourceBodyIds.Except(groups.SelectMany(group => group)).ToArray();
     if (missingBodyIds.Length != 0)
         failDisplay(new { missingBodyIds, error = "Body display inspection omits source CharacterBody locations" });
     var equipmentDefs = new System.Collections.Generic.HashSet<RoR2.EquipmentDef>(typedDefinitions.Values.OfType<RoR2.EquipmentDef>());
     var keyItemDefs = new System.Collections.Generic.HashSet<RoR2.KeyItemDef>(typedDefinitions.Values.OfType<RoR2.KeyItemDef>());
     System.Func<object, System.Type, UnityEngine.Object> loadDisplayAddress = (address, type) =>
     {
         if (!(address is string key) || !stringKeys.TryGetValue(key, out var candidates))
             return null;
         var assets = new System.Collections.Generic.HashSet<UnityEngine.Object>();
         foreach (var candidate in candidates)
         {
             var declaredType = System.Type.GetType((string)candidate["type"], false);
             if (declaredType == null || !type.IsAssignableFrom(declaredType))
                 continue;
             string owner = (string)candidate["bundle"];
             foreach (var dependency in manifest["bundles"][owner]["dependencies"])
                 openBundle((string)dependency);
             var asset = openBundle(owner).LoadAsset((string)candidate["path"], declaredType);
             if (asset)
                 assets.Add(asset);
         }
         return assets.Count == 1 ? assets.Single() : null;
     };
     foreach (var entry in typedBodies)
     {
         string path = entry.Key;
         var locator = entry.Value.GetComponent<RoR2.ModelLocator>();
         var target = locator ? locator.modelTransform : null;
         var model = target ? target.GetComponent<RoR2.CharacterModel>() : null;
         var ruleSet = model ? model.itemDisplayRuleSet : null;
         if (!ruleSet)
         {
             bodyDisplays.Add(new { path, ruleSet = (string)null, groups = 0, rules = 0 });
             continue;
         }
         var children = target.GetComponent<ChildLocator>();
         var displayGroups = ruleSet.keyAssetRuleGroups;
         if (displayGroups == null)
         {
             failDisplay(new { path, ruleSet = ruleSet.name, error = "Typed display rule groups are missing" });
             continue;
         }
         int rules = 0;
         var groupSlots = new System.Collections.Generic.List<object>();
         for (int i = 0; i < displayGroups.Length; i++)
         {
             var group = displayGroups[i];
             var key = group.keyAsset;
             var keyAddress = group.keyAssetAddress?.RuntimeKey;
             if (!key && (group.keyAssetAddress?.RuntimeKeyIsValid() ?? false))
                 key = loadDisplayAddress(keyAddress, typeof(UnityEngine.ScriptableObject));
             var registered = (key is RoR2.ItemDef item && itemDefs.Contains(item)) ||
                 (key is RoR2.EquipmentDef equipment && equipmentDefs.Contains(equipment)) ||
                 (key is RoR2.KeyItemDef keyItem && keyItemDefs.Contains(keyItem));
             var keyName = key ? key.name : null;
             var keyType = key ? key.GetType().FullName : null;
             if (!registered)
                 failDisplay(new { path, ruleSet = ruleSet.name, group = i, key = key ? key.name : null,
                     keyType, address = keyAddress, check = "key",
                     error = "Display key is outside the typed item, equipment and key item families" });
             var groupRules = group.displayRuleGroup.rules;
             if (groupRules == null)
             {
                 failDisplay(new { path, ruleSet = ruleSet.name, group = i, error = "Typed display rules are missing" });
                 continue;
             }
             rules += groupRules.Length;
             var ruleSlots = new System.Collections.Generic.List<object>();
             for (int j = 0; j < groupRules.Length; j++)
             {
                 var rule = groupRules[j];
                 var ruleType = rule.ruleType.ToString();
                 if (rule.ruleType != RoR2.ItemDisplayRuleType.ParentedPrefab)
                 {
                     ruleSlots.Add(new { rule = j, ruleType });
                     continue;
                 }
                 var follower = rule.followerPrefab;
                 var address = rule.followerPrefabAddress?.RuntimeKey;
                 if (!follower && (rule.followerPrefabAddress?.RuntimeKeyIsValid() ?? false))
                     follower = loadDisplayAddress(address, typeof(UnityEngine.GameObject)) as UnityEngine.GameObject;
                 var child = rule.childName;
                 var attachment = children ? children.FindChild(child) : null;
                 var followerName = follower ? follower.name : null;
                 ruleSlots.Add(new { rule = j, ruleType, follower = followerName, address, child,
                     attachment = describeTransform(attachment) });
                 if (follower)
                     displayPrefabs.Add(follower);
                 else
                     failDisplay(new { path, ruleSet = ruleSet.name, group = i, rule = j, key = keyName,
                         follower = followerName, address, child, check = "follower", error = "Typed display follower is missing" });
                 if (!attachment)
                     failDisplay(new { path, ruleSet = ruleSet.name, group = i, rule = j, key = keyName,
                         follower = followerName, address, child, check = "child", error = "Typed display attachment is missing" });
             }
             groupSlots.Add(new { group = i, key = keyName, keyType, keyAddress, registered, rules = ruleSlots });
         }
         bodyDisplays.Add(new { path, ruleSet = ruleSet.name, groups = displayGroups.Length, rules });
         displaySlots.Add(new { path, ruleSet = ruleSet.name, model = describeTransform(target), groups = groupSlots });
     }
     displaysComplete = bodyDisplays.Count == typedBodies.Count;
 }
}catch(System.Exception error){failures.Add(new{error=error.ToString()});}
finally{foreach(var bundle in bundles.Values.Reverse())bundle.Unload(true);}
